cmake_minimum_required(VERSION 3.20)
project(PixelPTX VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(pixelptx_core src/compiler.cpp)
target_include_directories(pixelptx_core PUBLIC include)
if(MSVC)
  target_compile_options(pixelptx_core PUBLIC /W4 /fp:strict)
else()
  target_compile_options(pixelptx_core PUBLIC -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
endif()
add_executable(pixelptx src/main.cpp)
target_link_libraries(pixelptx PRIVATE pixelptx_core)
option(PIXELPTX_ENABLE_CUDA "Build the optional NVIDIA CUDA Driver API runner" OFF)
if(PIXELPTX_ENABLE_CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(pixelptx_gpu src/gpu_main.cpp)
  target_link_libraries(pixelptx_gpu PRIVATE pixelptx_core CUDA::cuda_driver)
endif()
enable_testing()
add_executable(compiler_tests tests/compiler_tests.cpp)
target_link_libraries(compiler_tests PRIVATE pixelptx_core)
add_test(NAME compiler_correctness COMMAND compiler_tests)
